/**
 * Ltrust（エルトラスト）LINE公式アカウント　自動返信・リッチメニュー・予約
 *
 * スクリプトプロパティ：
 *   DATA_URL                     line_data.json の公開URL
 *   CHANNEL_ID / CHANNEL_SECRET  公式アカウント管理画面［設定］→［Messaging API］の値
 *   LINE_ID / DIAG_URL / BOOKING_URL / PHOTO_URL / MAP_FUKUI / MAP_HARUE / REVIEW_FUKUI / REVIEW_HARUE
 *   IMG_FUKUI / IMG_HARUE        リッチメニュー画像のURL
 *   TEST_MODE                    "off" にするまではテストモード（TEST_USER_IDS の人にだけ返信・メニュー表示）
 *   TEST_USER_IDS                テストする人のユーザーID（カンマ区切り）。LINEで「#myid」と送ると返ってくる
 *   SHEET_ID / CAL_FUKUI / CAL_HARUE   setupBooking() が自動で入れる
 *
 * 関数：
 *   setupBooking()   予約管理シートと店舗カレンダーを作り、毎時の処理を登録する（最初に1回）
 *   setupRichMenus() リッチメニューを作る／goLive() 全員に切り替え／rollback() 元に戻す
 *   refreshData()    line_data.json を読み直す／testRoute() 判定の確認
 */

function prop_(k) { return PropertiesService.getScriptProperties().getProperty(k) || ''; }
function setProp_(k, v) { PropertiesService.getScriptProperties().setProperty(k, v); }

/* ========== LINE の共通処理 ========== */

function data_() {
  const cache = CacheService.getScriptCache();
  const hit = cache.get('line_data');
  if (hit) return JSON.parse(hit);
  const res = UrlFetchApp.fetch(prop_('DATA_URL') + '?t=' + Date.now(), { muteHttpExceptions: true });
  if (res.getResponseCode() >= 300) throw new Error('DATA_URL を読めません: ' + res.getResponseCode());
  const txt = res.getContentText();
  try { cache.put('line_data', txt, 6 * 3600); } catch (e) {}
  return JSON.parse(txt);
}
function refreshData() { CacheService.getScriptCache().remove('line_data'); console.log(Object.keys(data_().messages)); }

function token_() {
  const fixed = prop_('CHANNEL_ACCESS_TOKEN');
  if (fixed) return fixed;
  const cache = CacheService.getScriptCache();
  const hit = cache.get('line_token');
  if (hit) return hit;
  const res = UrlFetchApp.fetch('https://api.line.me/oauth2/v3/token', {
    method: 'post',
    payload: { grant_type: 'client_credentials', client_id: prop_('CHANNEL_ID'), client_secret: prop_('CHANNEL_SECRET') },
    muteHttpExceptions: true
  });
  if (res.getResponseCode() >= 300) throw new Error('トークンの発行に失敗: ' + res.getContentText());
  const tk = JSON.parse(res.getContentText()).access_token;
  cache.put('line_token', tk, 13 * 60);
  return tk;
}

function fill_(obj) {
  return JSON.parse(JSON.stringify(obj).replace(/\{\{([A-Z_]+)\}\}/g, (_, k) => prop_(k)));
}

function api_(path, payload, method) {
  const res = UrlFetchApp.fetch('https://api.line.me' + path, {
    method: method || 'post', contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + token_() },
    payload: payload ? JSON.stringify(payload) : undefined, muteHttpExceptions: true
  });
  const code = res.getResponseCode();
  if (code >= 300) console.error(path, code, res.getContentText());
  return { code: code, body: res.getContentText() };
}
function reply_(token, messages) { return api_('/v2/bot/message/reply', { replyToken: token, messages: messages }); }
function push_(to, messages) { return api_('/v2/bot/message/push', { to: to, messages: messages }); }
function text_(s) { return { type: 'text', text: s }; }

function testers_() { return prop_('TEST_USER_IDS').split(',').map(x => x.trim()).filter(Boolean); }
function testMode_() { return prop_('TEST_MODE').toLowerCase() !== 'off'; }

/* ========== 受け口 ========== */

function doGet(e) {
  const p = (e && e.parameter) || {};
  let out;
  try {
    if (p.action === 'config') out = publicConfig_();
    else if (p.action === 'avail') out = avail_(p.store, Number(p.menu), p.from, Number(p.days || 14));
    else out = { ok: true };
  } catch (err) { out = { ok: false, error: String(err && err.message || err) }; }
  return json_(out);
}

function doPost(e) {
  const body = JSON.parse((e.postData && e.postData.contents) || '{}');
  if (body.events) {
    body.events.forEach(ev => { try { handle_(ev); } catch (err) { console.error(err); } });
    return ContentService.createTextOutput('ok');
  }
  let out;
  try {
    if (body.action === 'book') out = book_(body);
    else out = { ok: false, error: 'unknown action' };
  } catch (err) { out = { ok: false, error: String(err && err.message || err) }; }
  return json_(out);
}

function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

/* ========== LINE メッセージの判定 ========== */

function handle_(ev) {
  if (ev.type !== 'message' || ev.message.type !== 'text') return;
  const text = ev.message.text.trim();
  const userId = (ev.source && ev.source.userId) || '';
  if (text === '#myid') { reply_(ev.replyToken, [text_(userId)]); return; }
  if (testMode_() && testers_().indexOf(userId) < 0) return;

  const t = text.replace(/[Ａ-Ｚａ-ｚ０-９－]/g, c => String.fromCharCode(c.charCodeAt(0) - 0xFEE0)).replace(/\s/g, '').toUpperCase();
  let m;
  if ((m = t.match(/^予約確認(R-?[A-Z0-9]{4})$/))) { reply_(ev.replyToken, [text_(linkBooking_(normCode_(m[1]), userId))]); return; }
  if (t === '予約確認' || t === '予約の確認') { reply_(ev.replyToken, [text_(myBookings_(userId))]); return; }
  if ((m = t.match(/^キャンセル(R-?[A-Z0-9]{4})$/))) { reply_(ev.replyToken, [text_(cancelByUser_(normCode_(m[1]), userId))]); return; }

  const key = route_(text);
  if (!key) return;
  reply_(ev.replyToken, fill_(data_().messages[key]));
}

function normCode_(c) { c = c.replace('-', ''); return 'R-' + c.slice(1); }

function route_(text) {
  const t = text.replace(/[０-９]/g, c => String.fromCharCode(c.charCodeAt(0) - 0xFEE0)).replace(/\s/g, '');
  const m = t.match(/^(?:診断|タイプ|TYPE|type)?(\d{1,2})$/);
  if (m) { const n = Number(m[1]); return (n >= 1 && n <= 14) ? String(n) : 'fallback_help'; }
  if (t === 'お友達紹介' || t === 'お友だち紹介' || t === '紹介') return 'referral';
  if (/^診断/.test(t)) return 'fallback_help';
  return null;
}

  ['診断8', '８', 'タイプ14', '15', 'お友達紹介', 'こんにちは'].forEach(t => console.log(t, '→', route_(t)));
}

const SHEETS = {
  settings: '設定', hours: '営業時間', holidays: '休業日', menus: 'メニュー', bookings: '予約'
};
const BOOK_COLS = ['予約番号', '受付日時', '状態', '店舗', '日付', '開始', '終了', '番号', 'メニュー', '初回/2回目', '料金',
  'お名前', 'ふりがな', '電話', '紹介者', 'ご要望', 'LINEユーザーID', 'カレンダー予定ID', '出どころ', 'リマインド', 'フォロー'];

function ss_() { return SpreadsheetApp.openById(prop_('SHEET_ID')); }

function rows_(name) {
  const sh = ss_().getSheetByName(name);
  const v = sh.getDataRange().getDisplayValues();
  const head = v.shift();
  return v.map((r, i) => { const o = { _row: i + 2 }; head.forEach((h, j) => o[h] = r[j]); return o; });
}

function conf_() {
  const cache = CacheService.getScriptCache();
  const hit = cache.get('conf');
  if (hit) return JSON.parse(hit);
  const s = {};
  rows_(SHEETS.settings).forEach(r => s[r['項目']] = r['値']);
  const hours = { fukui: {}, harue: {} };
  rows_(SHEETS.hours).forEach(r => {
    const st = r['店舗'] === '春江店' ? 'harue' : (r['店舗'] === '福井店' ? 'fukui' : '');
    if (st && r['開始'] && r['終了']) hours[st][r['曜日']] = [r['開始'], r['終了']];
  });
  const holidays = rows_(SHEETS.holidays).filter(r => r['日付']).map(r => normDate_(r['日付']) + '|' + (r['店舗'] || ''));
  const menus = {};
  rows_(SHEETS.menus).forEach(r => {
    const n = Number(r['番号']); if (!n) return;
    menus[n] = { no: n, name: r['メニュー名'], min: Number(r['時間（分）']) || 0,
      first: num_(r['初回価格']), regular: num_(r['通常価格']), bookable: r['予約受付'] === '○', note: r['メモ'] || '' };
  });
  const c = {
    step: Number(s['枠の刻み（分）']) || 30,
    buffer: Number(s['施術の間隔（分）']) || 0,
    cutoffH: Number(s['受付締切（何時間前まで）']) || 24,
    maxDays: Number(s['何日先まで受け付けるか']) || 60,
    admins: String(s['通知先（LINEユーザーID）'] || '').split(',').map(x => x.trim()).filter(Boolean),
    remindHour: Number(s['前日リマインドの時刻']) || 18,
    digestHour: Number(s['毎晩の予約一覧の時刻']) || 21,
    followOn: String(s['翌日フォロー']).toUpperCase() === 'ON',
    followHour: Number(s['翌日フォローの時刻']) || 11,
    privateCals: String(s['私用カレンダーID（カンマ区切り）'] || '').split(',').map(x => x.trim()).filter(Boolean),
    hours: hours, holidays: holidays, menus: menus
  };
  cache.put('conf', JSON.stringify(c), 300);
  return c;
}
function clearConf() { CacheService.getScriptCache().remove('conf'); }

function num_(v) { return Number(String(v || '').replace(/[^\d]/g, '')) || 0; }
function normDate_(v) { const m = String(v).match(/(\d{4})\D(\d{1,2})\D(\d{1,2})/); return m ? m[1] + '-' + ('0' + m[2]).slice(-2) + '-' + ('0' + m[3]).slice(-2) : ''; }
function at_(dateStr, hhmm) { return new Date(dateStr + 'T' + (hhmm.length === 4 ? '0' + hhmm : hhmm) + ':00+09:00'); }
function fmt_(d, f) { return Utilities.formatDate(d, TZ, f); }
function wd_(dateStr) { return WD[Number(fmt_(at_(dateStr, '12:00'), 'u'))]; }
function jpDate_(dateStr) { const d = at_(dateStr, '12:00'); return fmt_(d, 'M月d日') + '（' + wd_(dateStr) + '）'; }
function yen_(n) { return String(n).replace(/\B(?=(\d{3})+(?!\d))/g, ',') + '円'; }

/* ========== 空き枠 ========== */

function publicConfig_() {
  const c = conf_();
  const menus = Object.keys(c.menus).map(k => c.menus[k]).filter(m => m.bookable)
    .map(m => ({ no: m.no, name: m.name, min: m.min, first: m.first, regular: m.regular, note: m.note }));
  const hours = {};
  Object.keys(c.hours).forEach(st => hours[st] = Object.keys(c.hours[st]));
  return { ok: true, menus: menus, stores: STORES, days: hours, maxDays: c.maxDays, cutoffH: c.cutoffH };
}

function busy_(from, to) {
  const c = conf_();
  const ids = [prop_('CAL_FUKUI'), prop_('CAL_HARUE')].concat(c.privateCals).filter(Boolean);
  const out = [];
  ids.forEach(id => {
    const cal = CalendarApp.getCalendarById(id);
    if (!cal) return;
    cal.getEvents(from, to).forEach(ev => {
      if (ev.isAllDayEvent()) out.push([ev.getAllDayStartDate().getTime(), ev.getAllDayEndDate().getTime(), true]);
      else out.push([ev.getStartTime().getTime(), ev.getEndTime().getTime(), false]);
    });
  });
  return out;
}

function slotsFor_(c, store, menu, dateStr, busy, now) {
  const m = c.menus[menu];
  if (!m || !m.bookable || !m.min) return [];
  const h = c.hours[store] && c.hours[store][wd_(dateStr)];
  if (!h) return [];
  if (c.holidays.indexOf(dateStr + '|') >= 0 || c.holidays.indexOf(dateStr + '|' + STORES[store]) >= 0) return [];
  const dayS = at_(dateStr, '00:00').getTime(), dayE = dayS + 864e5;
  if (busy.some(b => b[2] && b[0] < dayE && b[1] > dayS)) return []; // 終日の予定がある日は受け付けない
  const open = at_(dateStr, h[0]).getTime(), close = at_(dateStr, h[1]).getTime();
  const dur = m.min * 60e3, step = c.step * 60e3, buf = c.buffer * 60e3;
  const limit = now + c.cutoffH * 3600e3;
  const out = [];
  for (let s = open; s + dur <= close; s += step) {
    const e = s + dur;
    if (s < limit) continue;
    if (busy.some(b => !b[2] && s < b[1] + buf && e > b[0] - buf)) continue;
    out.push(fmt_(new Date(s), 'HH:mm'));
  }
  return out;
}

function avail_(store, menu, from, days) {
  if (!STORES[store]) throw new Error('店舗の指定がありません');
  const c = conf_();
  days = Math.min(Math.max(days || 14, 1), 31);
  const now = Date.now();
  const start = from ? normDate_(from) : fmt_(new Date(now), 'yyyy-MM-dd');
  const last = fmt_(new Date(now + c.maxDays * 864e5), 'yyyy-MM-dd');
  const busy = busy_(at_(start, '00:00'), new Date(at_(start, '00:00').getTime() + days * 864e5));
  const dates = {};
  for (let i = 0; i < days; i++) {
    const ds = fmt_(new Date(at_(start, '12:00').getTime() + i * 864e5), 'yyyy-MM-dd');
    if (ds > last) break;
    dates[ds] = slotsFor_(c, store, menu, ds, busy, now);
  }
  return { ok: true, store: store, menu: menu, dates: dates };
}

/* ========== 予約する ========== */

function book_(b) {
  if (b.website) return { ok: false, error: '送信できませんでした' }; // ボット対策
  const c = conf_();
  const store = b.store, menu = Number(b.menu), date = normDate_(b.date), time = String(b.time || '');
  const m = c.menus[menu];
  if (!STORES[store] || !m || !m.bookable || !date || !/^\d{2}:\d{2}$/.test(time)) return { ok: false, error: '予約内容が正しくありません' };
  const name = String(b.name || '').trim().slice(0, 40), kana = String(b.kana || '').trim().slice(0, 40);
  const tel = String(b.tel || '').replace(/[^\d]/g, '').slice(0, 15);
  if (!name || tel.length < 10) return { ok: false, error: 'お名前と電話番号を入力してください' };
  const visit = b.visit === 'repeat' ? '2回目以降' : '初回';
  const price = visit === '初回' ? (m.first || m.regular) : (m.regular || m.first);

  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const now = Date.now();
    const busy = busy_(at_(date, '00:00'), new Date(at_(date, '00:00').getTime() + 864e5));
    if (slotsFor_(c, store, menu, date, busy, now).indexOf(time) < 0) return { ok: false, error: 'taken' };
    const s = at_(date, time), e = new Date(s.getTime() + m.min * 60e3);
    const code = newCode_();
    const cal = CalendarApp.getCalendarById(prop_(store === 'fukui' ? 'CAL_FUKUI' : 'CAL_HARUE'));
    const desc = [
      '予約番号：' + code, 'メニュー：' + CIRC[menu - 1] + ' ' + m.name + '（' + m.min + '分）',
      visit + '　' + yen_(price), 'お名前：' + name + (kana ? '（' + kana + '）' : '') + ' 様',
      '電話：' + tel, b.referrer ? '紹介者：' + b.referrer : '', b.note ? 'ご要望：' + b.note : '',
      '受付：LINE予約フォーム'
    ].filter(Boolean).join('\n');
    const ev = cal.createEvent('【予約】' + name + '様　' + CIRC[menu - 1] + ' ' + m.name, s, e, { description: desc });
    const row = {
      '予約番号': code, '受付日時': fmt_(new Date(now), 'yyyy-MM-dd HH:mm'), '状態': '予約済', '店舗': STORES[store],
      '日付': date, '開始': time, '終了': fmt_(e, 'HH:mm'), '番号': String(menu), 'メニュー': m.name, '初回/2回目': visit,
      '料金': String(price), 'お名前': name, 'ふりがな': kana, '電話': tel, '紹介者': String(b.referrer || '').slice(0, 40),
      'ご要望': String(b.note || '').slice(0, 300), 'LINEユーザーID': '', 'カレンダー予定ID': ev.getId(),
      '出どころ': String(b.src || ''), 'リマインド': '', 'フォロー': ''
    };
    ss_().getSheetByName(SHEETS.bookings).appendRow(BOOK_COLS.map(k => row[k]));
    notifyAdmins_('【新しいご予約】' + STORES[store] + '\n' + jpDate_(date) + ' ' + time + '〜' + fmt_(e, 'HH:mm') +
      '\n' + CIRC[menu - 1] + ' ' + m.name + '（' + visit + ' ' + yen_(price) + '）\n' + name + ' 様　' + tel +
      (b.referrer ? '\n紹介者：' + b.referrer : '') + (b.note ? '\nご要望：' + b.note : '') + '\n予約番号：' + code);
    return { ok: true, code: code, store: STORES[store], date: jpDate_(date), time: time + '〜' + fmt_(e, 'HH:mm'),
      menu: CIRC[menu - 1] + ' ' + m.name, price: yen_(price), visit: visit, lineId: prop_('LINE_ID') };
  } finally { lock.releaseLock(); }
}

function newCode_() {
  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  const used = rows_(SHEETS.bookings).map(r => r['予約番号']);
  let c;
  do { c = 'R-'; for (let i = 0; i < 4; i++) c += chars[Math.floor(Math.random() * chars.length)]; } while (used.indexOf(c) >= 0);
  return c;
}

function findBooking_(code) { return rows_(SHEETS.bookings).filter(r => r['予約番号'] === code)[0]; }
function setCell_(row, col, value) {
  ss_().getSheetByName(SHEETS.bookings).getRange(row, BOOK_COLS.indexOf(col) + 1).setValue(value);
}
function storeKey_(name) { return name === '春江店' ? 'harue' : 'fukui'; }

function bookingText_(r) {
  return '予約番号：' + r['予約番号'] + '\n日時：' + jpDate_(r['日付']) + ' ' + r['開始'] + '〜' + r['終了'] +
    '\n店舗：' + r['店舗'] + '\nメニュー：' + CIRC[Number(r['番号']) - 1] + ' ' + r['メニュー'] +
    '（' + r['初回/2回目'] + ' ' + yen_(r['料金']) + '）\nお名前：' + r['お名前'] + ' 様';
}

function linkBooking_(code, userId) {
  const r = findBooking_(code);
  if (!r) return '予約番号 ' + code + ' が見つかりませんでした。お手数ですが、このトークでお知らせください。';
  if (r['LINEユーザーID'] && r['LINEユーザーID'] !== userId) return 'この予約番号は、すでに別のLINEで登録されています。';
  if (!r['LINEユーザーID']) setCell_(r._row, 'LINEユーザーID', userId);
  const key = storeKey_(r['店舗']);
  return 'ご予約ありがとうございます。\n\n' + bookingText_(r) +
    '\n\n前日の夕方に、このLINEでお知らせします。\n変更やキャンセルは「キャンセル ' + code + '」と送るか、このトークでご連絡ください。' +
    '\n\n' + r['店舗'] + 'の地図：' + prop_(key === 'harue' ? 'MAP_HARUE' : 'MAP_FUKUI');
}

function myBookings_(userId) {
  const today = fmt_(new Date(), 'yyyy-MM-dd');
  const list = rows_(SHEETS.bookings).filter(r => r['LINEユーザーID'] === userId && r['状態'] === '予約済' && r['日付'] >= today);
  if (!list.length) return '今後のご予約は見つかりませんでした。ご予約は下のメニューの「はじめての予約」「2回目以降の予約」からどうぞ。';
  return 'ご予約の内容です。\n\n' + list.map(bookingText_).join('\n\n');
}

function cancelByUser_(code, userId) {
  const r = findBooking_(code);
  if (!r || r['LINEユーザーID'] !== userId) return '予約番号 ' + code + ' のご予約が、このLINEでは見つかりませんでした。このトークでお知らせください。';
  if (r['状態'] !== '予約済') return 'このご予約は、すでにキャンセル済みです。';
  const c = conf_();
  if (at_(r['日付'], r['開始']).getTime() - Date.now() < c.cutoffH * 3600e3)
    return '直前のためLINEからはキャンセルできません。お手数ですが、このトークでご連絡ください。';
  cancelBooking_(r, 'お客様がLINEでキャンセル');
  return 'キャンセルを承りました。\n\n' + bookingText_(r) + '\n\nまたのご予約をお待ちしています。';
}

function cancelBooking_(r, reason) {
  try {
    const cal = CalendarApp.getCalendarById(prop_(storeKey_(r['店舗']) === 'harue' ? 'CAL_HARUE' : 'CAL_FUKUI'));
    const ev = cal.getEventById(r['カレンダー予定ID']);
    if (ev) ev.deleteEvent();
  } catch (e) { console.error(e); }
  setCell_(r._row, '状態', 'キャンセル');
  notifyAdmins_('【キャンセル】' + reason + '\n' + bookingText_(r));
}

function notifyAdmins_(msg) {
  conf_().admins.forEach(id => push_(id, [text_(msg)]));
}

/* ========== 毎時の処理（リマインド・毎晩の一覧・翌日フォロー） ========== */

function hourly() {
  const c = conf_();
  const hour = Number(fmt_(new Date(), 'H'));
  if (hour === c.remindHour) sendReminders_();
  if (hour === c.digestHour) sendDigest_();
  if (c.followOn && hour === c.followHour) sendFollowUps_();
}

function sendReminders_() {

function sendDigest_() {
  const today = fmt_(new Date(), 'yyyy-MM-dd');
  const tomorrow = fmt_(new Date(Date.now() + 864e5), 'yyyy-MM-dd');
  const all = rows_(SHEETS.bookings);
  const newToday = all.filter(r => String(r['受付日時']).indexOf(today) === 0);
  const tmr = all.filter(r => r['日付'] === tomorrow && r['状態'] === '予約済').sort((a, b) => a['開始'] < b['開始'] ? -1 : 1);
  if (!newToday.length && !tmr.length) return;
  const lines = ['【予約のまとめ】' + jpDate_(today)];
  lines.push('', '■ 今日入った予約・キャンセル：' + newToday.length + '件');
  newToday.forEach(r => lines.push((r['状態'] === 'キャンセル' ? '×' : '・') + jpDate_(r['日付']) + ' ' + r['開始'] + ' ' + r['店舗'] + ' ' + r['お名前'] + '様 ' + CIRC[Number(r['番号']) - 1]));
  lines.push('', '■ 明日の予定：' + tmr.length + '件');
  tmr.forEach(r => lines.push('・' + r['開始'] + '〜' + r['終了'] + ' ' + r['店舗'] + ' ' + r['お名前'] + '様 ' + CIRC[Number(r['番号']) - 1] + ' ' + r['メニュー']));
  lines.push('', 'タイムツリーへの入力はこの一覧を見ながらどうぞ。');
  notifyAdmins_(lines.join('\n'));
}

/* ========== リッチメニュー ========== */

function setupRichMenus() {
  ['rm-fukui', 'rm-harue'].forEach(a => api_('/v2/bot/richmenu/alias/' + a, null, 'delete'));
  const old = [prop_('RICHMENU_FUKUI'), prop_('RICHMENU_HARUE')].filter(Boolean);
  const ids = {};
  ['fukui', 'harue'].forEach(store => {
    const r = api_('/v2/bot/richmenu', fill_(data_().richmenus[store]));
    if (r.code >= 300) throw new Error('リッチメニューの作成に失敗: ' + r.body);
    const id = JSON.parse(r.body).richMenuId;
    const blob = UrlFetchApp.fetch(prop_(store === 'fukui' ? 'IMG_FUKUI' : 'IMG_HARUE')).getBlob();
    const up = UrlFetchApp.fetch('https://api-data.line.me/v2/bot/richmenu/' + id + '/content', {
      method: 'post', contentType: 'image/jpeg', headers: { Authorization: 'Bearer ' + token_() },
      payload: blob.getBytes(), muteHttpExceptions: true
    });
    if (up.getResponseCode() >= 300) throw new Error('画像のアップロードに失敗: ' + up.getContentText());
    api_('/v2/bot/richmenu/alias', { richMenuAliasId: 'rm-' + store, richMenuId: id });
    ids[store] = id;
  });
  setProp_('RICHMENU_FUKUI', ids.fukui);
  setProp_('RICHMENU_HARUE', ids.harue);
  if (testMode_()) {
    testers_().forEach(u => api_('/v2/bot/user/' + u + '/richmenu/' + ids.fukui, null, 'post'));
    console.log('テストする人にだけ表示しました', ids, testers_());
  } else {
    api_('/v2/bot/user/all/richmenu/' + ids.fukui, null, 'post');
    console.log('全員に表示しました', ids);
  }
  old.forEach(id => api_('/v2/bot/richmenu/' + id, null, 'delete')); // 前の版を片付ける
}

function goLive() {
  const id = prop_('RICHMENU_FUKUI');
  if (!id) throw new Error('先に setupRichMenus() を実行してください');
  api_('/v2/bot/user/all/richmenu/' + id, null, 'post');
  setProp_('TEST_MODE', 'off');
  console.log('全員に切り替えました。あいさつメッセージの差し替えとお知らせ配信を忘れずに');
}

function rollback() {
  ['rm-fukui', 'rm-harue'].forEach(a => api_('/v2/bot/richmenu/alias/' + a, null, 'delete'));
  api_('/v2/bot/user/all/richmenu', null, 'delete');
  ['RICHMENU_FUKUI', 'RICHMENU_HARUE'].forEach(k => { const id = prop_(k); if (id) api_('/v2/bot/richmenu/' + id, null, 'delete'); });
  setProp_('TEST_MODE', 'on');
  console.log('元のメニューに戻しました。自動返信はテストする人だけです');
}
